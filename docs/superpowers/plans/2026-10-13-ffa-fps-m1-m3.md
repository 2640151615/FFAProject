# FFA 竞技射击游戏 M1–M3 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把空的 Rojo 工程做成一个可玩的单枪混战：玩家第一人称出生、开枪、命中扣血、死亡重生，且有视图模型、后坐力、音效与准星。

**Architecture:** 服务器权威。客户端只发"开火意图"（朝向 + 时间戳），服务器做位置回溯 + raycast 判定伤害。纯逻辑（武器表、伤害、限速、回溯插值）抽成不依赖引擎的模块，用 Lune 做单元测试；引擎相关部分用 Roblox Studio MCP 实机验证。

**Tech Stack:** Luau、Rojo 7.7.1、Lune（单元测试）、Roblox Studio（实机验证）

**Spec:** `docs/superpowers/specs/2026-10-13-ffa-fps-design.md`

## Global Constraints

以下为全项目恒定约束，每个任务都隐含包含，不再逐条重复：

- **纯第一人称**：任何时候不出现自己角色的第三人称视图
- **R6 骨架**：由 Studio 的 `Game Settings → Avatar → Rig Type` 控制，**代码不控制**；R6 部件为 `Head` / `Torso` / `Left Arm` / `Right Arm` / `Left Leg` / `Right Leg`
- **服务器权威**：客户端上报的任何伤害数字一律不采信
- **竞技向**：首发零扩散（`firstShotSpread = 0`），扩散只在连发时累积
- **移动端一等公民**：所有输入必须经 `Input` 抽象层，业务代码不直接读鼠标
- **数值常量**（写入 `Config.luau`）：`RespawnDelay = 3`、`ScoreLimit = 30`、`MaxRequestsPerSecond = 20`、`RequestBurst = 40`、`MaxPlayers = 12`、`SpawnProtectionTime = 2`
- **出生保护**：短暂无敌，**开火或移动即失效**
- **无击杀回放**
- **脚步空间音效**为设计需求，非可选
- 所有文件位于 `src/{server,client,shared}`，由 Rojo 同步
- 注释用中文，标识符用英文

## Review Focus

以下是 spec 暗示、但任务测试未直接覆盖、且最可能坑到真人的输入或情形。每一条都在所属任务里配了测试：

1. **weaponId 为 nil 或垃圾字符串**——`Weapons.get` 必须返回 nil 而不是报错（远程请求可携带任意字符串）
2. **限速器在被长时间闲置后**——令牌必须封顶于 `burst`，不能无限累积成一次巨量放行
3. **同一时间戳被回溯两次**——位置回溯必须返回同一结果，不得因读取顺序改变
4. **玩家在开火瞬间死亡的边界**——服务器不得对已死玩家结算伤害
5. **partName 不在 R6 部件表里**（例如未来改用 R15 或命中装饰物）——伤害计算必须回退到基础伤害，而不是返回 nil 导致扣血为 NaN

这个文件是基础设施，接口必须精确，所以给出完整实现：

```lua
-- 极简断言骨架。只做三件事：收集用例、跑、汇总退出码。
-- 刻意不引第三方框架：回归防线不该押在小众包的维护状况上。

local harness = {}

local suites = {}
local currentSuite = nil
local failures = {}
local passed = 0

function harness.describe(name, fn)
	currentSuite = { name = name, cases = {} }
	table.insert(suites, currentSuite)
	fn()
	currentSuite = nil
end

function harness.it(name, fn)
	assert(currentSuite, "it() 必须写在 describe() 内部")
	table.insert(currentSuite.cases, { name = name, fn = fn })
end

local expectValue = {}
expectValue.__index = expectValue

local function fail(msg)
	error(msg, 3)
end

function expectValue:toBe(expected)
	if self.value ~= expected then
		fail(string.format("期望 %s，实际 %s", tostring(expected), tostring(self.value)))
	end
end

function expectValue:toEqual(expected)
	-- 深比较，顺序无关地比较表键
	local function deepEq(a, b)
		if type(a) ~= type(b) then return false end
		if type(a) ~= "table" then return a == b end
		for k, v in pairs(a) do
			if not deepEq(v, b[k]) then return false end
		end
		for k in pairs(b) do
			if a[k] == nil then return false end
		end
		return true
	end
	if not deepEq(self.value, expected) then
		fail("深比较不相等")
	end
end

function expectValue:toBeTruthy()
	if not self.value then fail("期望真值，实际 " .. tostring(self.value)) end
end

function expectValue:toThrow()
	if type(self.value) ~= "function" then fail("toThrow 需要一个函数") end
	local ok = pcall(self.value)
	if ok then fail("期望抛出错误，但没有") end
end

function harness.expect(value)
	return setmetatable({ value = value }, expectValue)
end

function harness.run()
	for _, suite in ipairs(suites) do
		for _, case in ipairs(suite.cases) do
			local ok, err = pcall(case.fn)
			if ok then
				passed += 1
			else
				table.insert(failures, string.format("%s > %s\n    %s", suite.name, case.name, tostring(err)))
			end
		end
	end

	print(string.format("\n%d 通过，%d 失败\n", passed, #failures))
	for _, f in ipairs(failures) do
		print("失败: " .. f)
	end

	if #failures > 0 then
		process.exit(1)
	end
end

return harness
```

## Task 12: 触屏弱辅助瞄准（M3）

**Files:**
- Create: `src/client/AimAssist.luau`
- Create: `test/aimassist.spec.luau`

**Interfaces:**
- Produces: `AimAssist.new(maxAngleDegrees: number, strength: number) -> Assist`
- Produces: `Assist:apply(aimDirection: Vector3, candidates: {Vector3}, isAiming: boolean) -> Vector3`

- [ ] **Step 1: 写失败测试 `test/aimassist.spec.luau`**

本任务最重要的不是功能，而是**给"弱"这个字上一道锁**——spec 的反目标写死"不要变成自动瞄准大战"。所以测试直接钉住角度上限：

```lua
local harness = require("./harness")
local describe, it, expect = harness.describe, harness.it, harness.expect
local AimAssist = require("../src/client/AimAssist")

-- 返回两个单位向量的夹角（度）
local function angleBetween(a, b)
	local cos = math.clamp(a.Unit:Dot(b.Unit), -1, 1)
	return math.deg(math.acos(cos))
end

describe("AimAssist", function()
	it("没有候选目标时方向不变", function()
		local a = AimAssist.new(3, 0.5)
		local aim = Vector3.new(0, 0, -1)
		local out = a:apply(aim, {}, false)
		expect(out):toEqual(aim)
	end)

	it("偏移永远不超过角度上限（不许自动跟枪）", function()
		local a = AimAssist.new(3, 1.0)
		local aim = Vector3.new(0, 0, -1)
		-- 目标偏离 90 度：一个真正"自动瞄准"的实现会直接吸过去
		local far = Vector3.new(-1, 0, 0)
		local out = a:apply(aim, { far }, false)
		expect(angleBetween(aim, out) <= 3.001):toBe(true)
	end)

	it("开镜时辅助更弱", function()
		local a = AimAssist.new(3, 1.0)
		local aim = Vector3.new(0, 0, -1)
		local near = Vector3.new(0.1, 0, -1)
		local hip = a:apply(aim, { near }, false)
		local ads = a:apply(aim, { near }, true)
		expect(angleBetween(aim, ads) <= angleBetween(aim, hip) + 1e-6):toBe(true)
	end)

	it("strength 为 0 时完全不动", function()
		local a = AimAssist.new(3, 0)
		local aim = Vector3.new(0, 0, -1)
		expect(a:apply(aim, { Vector3.new(0.1, 0, -1) }, false)):toEqual(aim)
	end)
end)

harness.run()
```
